(* Resumo da sintaxe da extensão, incluindo listas genéricas (06/10/2026).
   Gramática completa e executável: src/main/javacc/Functional1.jj.
   Expressao inclui os operadores, let e if da Funcional 1 original.
   Funções preservam a declaração: let fun f x y = Expressao in Expressao.
   A etapa 2 infere parâmetros de registros/listas sem novas anotações sintáticas.
*)

DeclRegistro        ::= "let" "record" ID "{" [ DeclCampos ] "}" "in" Expressao ;
DeclCampos          ::= DeclCampo { "," DeclCampo } ;
DeclCampo           ::= ID ":" TipoCampo ;
TipoCampo           ::= "Int" | "String" | "Boolean" | ID | "[" TipoCampo "]" ;

InstanciacaoRegistro ::= ID "{" [ AtribuicaoCampos ] "}" ;
AtribuicaoCampos    ::= AtribuicaoCampo { "," AtribuicaoCampo } ;
AtribuicaoCampo     ::= ID ":" Expressao ;

LiteralLista        ::= "[" [ Expressao { "," Expressao } ] "]" ;
OperacaoLista       ::= ( "head" | "tail" | "isEmpty" ) "(" Expressao ")" ;

Primaria            ::= Atomo { "." ID } ;
Atomo               ::= ID | ValorPrimitivo | "(" Expressao ")"
                      | ChamadaFuncao | InstanciacaoRegistro
                      | LiteralLista | OperacaoLista ;
ChamadaFuncao       ::= ID "(" Expressao { "," Expressao } ")" ;

(* ID e ValorPrimitivo usam o léxico original do arquivo .jj.
   record, head, tail e isEmpty são palavras reservadas.
   DeclRegistro é uma alternativa de expressão, como let e if.
   TipoCampo aceita registros nomeados no escopo e listas aninhadas.
   Listas têm elementos de tipos compatíveis; não são restritas a registros.
   Regras semânticas e limitações: ../docs/IMPLEMENTACAO.md.
*)
